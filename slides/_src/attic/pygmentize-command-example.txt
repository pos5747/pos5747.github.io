pygmentize -l r -f html -O full,style=friendly_grayscale -o ~/Downloads/script.html <<EOF
# load packages
library(tidyverse)
library(sandwich)

# simulate fake data
n <- 100
m <- 2
b <- 10
x <- runif(n, -1, 1)
y <- m*x + b + rnorm(n)
df <- data.frame(x, y)

# fit model
fit <- lm(y ~ x, data = df)

# print estimates
coef(fit)
vcovHC(fit, type = "HC2")

# plot
ggplot(df, aes(x = x, y = y)) + 
  geom_point()
EOF

open ~/Downloads/script.html
